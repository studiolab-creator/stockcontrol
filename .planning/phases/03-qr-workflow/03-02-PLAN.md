```typescript
'use client'
import { useEffect, useRef, useState } from 'react'
import { useRouter } from 'next/navigation'

// UUID v4 regex: 8-4-4-4-12 hex chars with hyphens
const UUID_RE = /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i

export function QrScannerClient() {
  const videoRef = useRef<HTMLVideoElement>(null)
  const scannerRef = useRef<import('qr-scanner').default | null>(null)
  const router = useRouter()
  const [httpsError, setHttpsError] = useState(false)

  useEffect(() => {
    // HTTPS check — getUserMedia fails on non-HTTPS non-localhost (RESEARCH.md Pitfall 6, CLAUDE.md constraint)
    const isSecure =
      location.protocol === 'https:' || location.hostname === 'localhost'
    if (!isSecure) {
      setHttpsError(true)
      return
    }

    if (!videoRef.current) return

    // Dynamic import inside useEffect — second SSR safety layer.
    // Primary guard is ssr: false in escanear/page.tsx (RESEARCH.md Pitfall 1).
    // The import() here also prevents the module from loading on any non-browser render.
    import('qr-scanner').then(({ default: QrScanner }) => {
      const scanner = new QrScanner(
        videoRef.current!,
        (result) => {
          const uuid = result.data
          // Only navigate for valid UUID v4 — prevents acting on arbitrary QR codes
          // D-05: on successful scan, auto-navigate to /reducir/[productId]
          if (UUID_RE.test(uuid)) {
            scanner.stop()
            router.push(`/reducir/${uuid}`)
          }
        },
        {
          returnDetailedScanResult: true,     // result.data instead of deprecated string arg
          preferredCamera: 'environment',     // rear camera on mobile
          onDecodeError: (err) => {
            // Fires every frame when no QR found — this is normal, not an error (RESEARCH.md Pitfall 3)
            if (err === QrScanner.NO_QR_CODE_FOUND) return
            console.error('QR decode error:', err)
          },
        },
      )
      scannerRef.current = scanner
      scanner.start().catch((err) => {
        console.error('Camera start failed:', err)
      })
    })

    // Cleanup: destroy stops the camera stream and terminates the Web Worker.
    // Required to prevent battery drain after navigation (RESEARCH.md Pitfall 1).
    return () => {
      scannerRef.current?.destroy()
      scannerRef.current = null
    }
  }, [router])

  if (httpsError) {
    return (
      <div className="flex flex-col items-center justify-center py-16 text-center">
        <p className="text-base font-semibold text-foreground mb-1">
          Se requiere conexión segura
        </p>
        <p className="text-sm text-muted-foreground">
          La cámara solo funciona en HTTPS. Accedé desde la URL segura de la aplicación.
        </p>
      </div>
    )
  }

  return (
    <div className="flex flex-col gap-4">
      <p className="text-sm text-muted-foreground">
        Apuntá la cámara al código QR del producto.
      </p>
      <div className="w-full max-w-sm mx-auto rounded-lg overflow-hidden border border-border">
        <video ref={videoRef} className="w-full" />
      </div>
    </div>
  )
}
```

**File 2: `src/app/(app)/escanear/page.tsx`**

```typescript
import dynamic from 'next/dynamic'
import { getAuthenticatedUser } from '@/lib/dal'

// CRITICAL: ssr: false prevents qr-scanner's Web Worker from being evaluated during SSR.
// The QrScannerClient component uses browser APIs (navigator.mediaDevices, Worker).
// Without this, Next.js will throw "ReferenceError: Worker is not defined" during SSR.
// (RESEARCH.md Pitfall 1 / CLAUDE.md architecture constraint for QR scanning)
const QrScannerClient = dynamic(
  () =>
    import('@/components/qr-scanner-client').then((m) => m.QrScannerClient),
  {
    ssr: false,
    loading: () => (
      <p className="text-sm text-muted-foreground">Cargando cámara...</p>
    ),
  },
)

export default async function EscanearPage() {
  // Any authenticated user (ADMIN or OPERADOR) can scan QR codes
  await getAuthenticatedUser()

  return (
    <div>
      <h1 className="text-xl font-semibold text-foreground mb-6">Escanear QR</h1>
      <QrScannerClient />
    </div>
  )
}
```

Key rules:
- `dynamic(..., { ssr: false })` MUST be in the page file (Server Component), not in the component file itself.
- `getAuthenticatedUser()` (not `requireAdmin()`) — both ADMIN and OPERADOR can scan.
- The UUID regex `/^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i` is stricter than the PATTERNS.md version (which used `[0-9a-f-]{36}` — too permissive, matches strings with wrong hyphen placement).

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Browser camera → QR decoder | Decoded QR content is untrusted input from any physical QR code |
| QR decoded string → router.push | Must validate UUID format before navigating to prevent path injection |
| getUserMedia → browser | HTTPS required; non-HTTPS must be detected and shown as error |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-03-02-01 | Spoofing | Camera permission on non-HTTPS | mitigate | `location.protocol !== 'https:' && location.hostname !== 'localhost'` check in useEffect; renders error UI instead of starting scanner (CLAUDE.md HTTPS constraint) |
| T-03-02-02 | Tampering | Crafted QR with arbitrary path string | mitigate | UUID_RE regex validation before `router.push` — only UUID v4 format strings trigger navigation; non-UUID QR codes are ignored |
| T-03-02-03 | Denial of Service | Camera stream not cleaned up | mitigate | `scanner.destroy()` called in useEffect cleanup — stops stream and terminates Web Worker on unmount |
| T-03-02-04 | Elevation of Privilege | /escanear page access | accept | `getAuthenticatedUser()` redirects unauthenticated users to /login; both ADMIN and OPERADOR are permitted to scan (per design) |
| T-03-02-05 | Information Disclosure | Nav items visible regardless of role | accept | /qr page calls `requireAdmin()` independently — nav item visibility does not grant access; OPERADOR hitting /qr gets redirected to /dashboard |
</threat_model>
